from core.db.mixins import TimeStampedModel, UserScopedModel

class LeaveType(models.Model):

    days_per_year = models.IntegerField()
    is_half_day = models.BooleanField(default=False)

    def __str__(self):
        return self.name

class LeaveBalance(models.Model):
    user = models.ForeignKey('auth.User', on_delete=models.CASCADE)
    leave_type = models.ForeignKey(LeaveType, on_delete=models.CASCADE)
    balance = models.DecimalField(max_digits=10, decimal_places=2, default=0)

    class Meta:
        unique_together = ('user', 'leave_type')

    def __str__(self):
        return f"{self.user.username} - {self.leave_type.name} Balance"

class LeaveRequest(models.Model):
    STATUS_CHOICES = [
        ('DRAFT', 'Draft'),
        ('PENDING_APPROVAL', 'Pending Approval'),
        ('RECOMMENDED', 'Recommended'),
        ('APPROVED', 'Approved'),
        ('REJECTED', 'Rejected'),
        ('CANCELLED', 'Cancelled'),
    ]
    user = models.ForeignKey('auth.User', on_delete=models.CASCADE)
    leave_type = models.ForeignKey(LeaveType, on_delete=models.CASCADE)
    start_date = models.DateField()
    end_date = models.DateField()
    is_station_leave = models.BooleanField(default=False)
    station_start_date = models.DateField(null=True, blank=True)
    station_end_date = models.DateField(null=True, blank=True)
    station_address = models.CharField(max_length=255, null=True, blank=True)
    emergency_contact_no = models.CharField(max_length=15, null=True, blank=True)
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default='DRAFT')
    remarks = models.TextField(null=True, blank=True)
    medical_certificate = models.FileField(upload_to='leave_medical_certificates/', null=True, blank=True)
    created_by = models.ForeignKey('auth.User', related_name='created_leave_requests', on_delete=models.CASCADE)
    updated_by = models.ForeignKey('auth.User', related_name='updated_leave_requests', on_delete=models.CASCADE)

    class Meta:
        ordering = ('-created_at',)

    def __str__(self):
        return f"{self.user.username} - {self.leave_type.name} - {self.status}"

from rest_framework import serializers
from .models import LeaveType, LeaveBalance, LeaveRequest
from django.utils import timezone

class LeaveTypeSerializer(serializers.ModelSerializer):
    class Meta:
        model = LeaveType
        fields = '__all__'

class LeaveBalanceSerializer(serializers.ModelSerializer):
    class Meta:
        model = LeaveBalance
        fields = '__all__'

class LeaveRequestSerializer(serializers.ModelSerializer):
    leave_type = LeaveTypeSerializer()
    user = serializers.PrimaryKeyRelatedField(queryset=User.objects.all())
    created_by = serializers.PrimaryKeyRelatedField(queryset=User.objects.all())
    updated_by = serializers.PrimaryKeyRelatedField(queryset=User.objects.all())

    class Meta:
        model = LeaveRequest
        fields = '__all__'

    def validate(self, data):
        if data['is_station_leave']:
            if not data['station_start_date'] or not data['station_end_date']:
                raise serializers.ValidationError("Station leave requires start and end dates.")
        if data['start_date'] > data['end_date']:
            raise serializers.ValidationError("Start date must be before end date.")
        return data

    def validate_leave_type(self, value):
        if value.days_per_year == 0:
            raise serializers.ValidationError("Invalid leave type.")
        return value

    def validate_station_leave(self, value):
        if value and not self.instance:
            raise serializers.ValidationError("Station leave cannot be added to an existing request.")
        return value

    def validate_medical_certificate(self, value):
        if self.instance and self.instance.leave_type.name == 'COL' and not value:
            raise serializers.ValidationError("Medical certificate is required for COL leave.")
        return value

    def create(self, validated_data):
        leave_type_data = validated_data.pop('leave_type')
        leave_type = LeaveType.objects.get(id=leave_type_data['id'])
        validated_data['leave_type'] = leave_type
        validated_data['created_by'] = self.context['request'].user
        validated_data['updated_by'] = self.context['request'].user
        leave_request = LeaveRequest.objects.create(**validated_data)
        leave_request.update_balance()
        return leave_request

    def update(self, instance, validated_data):
        leave_type_data = validated_data.pop('leave_type')
        leave_type = LeaveType.objects.get(id=leave_type_data['id'])
        validated_data['leave_type'] = leave_type
        validated_data['updated_by'] = self.context['request'].user
        leave_request = super().update(instance, validated_data)
        leave_request.update_balance()
        return leave_request
